1. Bit Stuffing

#include <stdio.h>
#include <string.h>

int main(void) {
    char data[100], out[220];
    int i, ones = 0, k = 0;

    printf("Enter bit string: ");
    scanf("%99s", data);

    strcpy(out, "01111110");
    k = (int)strlen(out);

    for (i = 0; data[i] != '\0'; i++) {
        out[k++] = data[i];

        if (data[i] == '1') {
            ones++;
            if (ones == 5) {
                out[k++] = '0';
                ones = 0;
            }
        } else {
            ones = 0;
        }
    }

    strcpy(out + k, "01111110");
    printf("After bit stuffing: %s\\n", out);
    return 0;
}

2. Character Stuffing

#include <stdio.h>
#include <string.h>

int main(void) {
    char data[100], out[220], start, end;
    int i, k = 0;

    printf("Enter data (no spaces): ");
    scanf("%99s", data);
    printf("Enter starting delimiter: ");
    scanf(" %c", &start);
    printf("Enter ending delimiter: ");
    scanf(" %c", &end);

    out[k++] = start;
    for (i = 0; data[i] != '\0'; i++) {
        out[k++] = data[i];
        if (data[i] == start || data[i] == end)
            out[k++] = data[i];
    }
    out[k++] = end;
    out[k] = '\0';

    printf("After character stuffing: %s\\n", out);
    return 0;
}

3. Distance Vector Routing Algorithm

Enter a cost matrix. Use 0 on the diagonal and 999 where there is no direct link. This is a simplified distance-vector-style shortest-path demonstration.

#include <stdio.h>

#define MAX 10
#define INF 999

int main(void) {
    int cost[MAX][MAX], dist[MAX][MAX], n, i, j, k, changed;

    printf("Enter number of nodes (max %d): ", MAX);
    scanf("%d", &n);

    printf("Enter cost matrix (use %d for no direct link):\\n", INF);
    for (i = 0; i < n; i++) {
        for (j = 0; j < n; j++) {
            scanf("%d", &cost[i][j]);
            dist[i][j] = cost[i][j];
        }
    }

    do {
        changed = 0;
        for (i = 0; i < n; i++) {
            for (j = 0; j < n; j++) {
                for (k = 0; k < n; k++) {
                    if (cost[i][k] != INF && dist[k][j] != INF &&
                        dist[i][j] > cost[i][k] + dist[k][j]) {
                        dist[i][j] = cost[i][k] + dist[k][j];
                        changed = 1;
                    }
                }
            }
        }
    } while (changed);

    printf("\\nFinal distance table:\\n");
    for (i = 0; i < n; i++) {
        printf("Router %d: ", i);
        for (j = 0; j < n; j++) {
            if (dist[i][j] >= INF) printf("INF ");
            else printf("%d ", dist[i][j]);
        }
        printf("\\n");
    }
    return 0;
}

4. CRC-CCITT (with error detection)

Uses CRC-16/CCITT-FALSE (polynomial 0x1021, initial value 0xFFFF) on text bytes. This is a byte-oriented CRC demo, not a bit-string CRC lab variant.

#include <stdio.h>
#include <string.h>
#include <stdint.h>

uint16_t crc16(const unsigned char *data, size_t len) {
    uint16_t crc = 0xFFFF;
    size_t i;
    int bit;

    for (i = 0; i < len; i++) {
        crc ^= (uint16_t)data[i] << 8;
        for (bit = 0; bit < 8; bit++)
            crc = (crc & 0x8000) ? (uint16_t)((crc << 1) ^ 0x1021)
                                 : (uint16_t)(crc << 1);
    }
    return crc;
}

int main(void) {
    char data[100], frame[110];
    uint16_t crc;
    int choice, pos;
    size_t n, i;

    printf("Enter data (text, no spaces): ");
    scanf("%99s", data);
    n = strlen(data);
    crc = crc16((unsigned char *)data, n);

    for (i = 0; i < n; i++) frame[i] = data[i];
    frame[n] = (char)(crc >> 8);
    frame[n + 1] = (char)(crc & 0xFF);

    printf("CRC-CCITT: %04X\\n", crc);
    printf("Frame data followed by CRC bytes: ");
    for (i = 0; i < n + 2; i++) printf("%02X ", (unsigned char)frame[i]);
    printf("\\nInject an error? 0=No, 1=Yes: ");
    scanf("%d", &choice);

    if (choice == 1) {
        printf("Enter bit position to flip (1 to %d): ", (int)((n + 2) * 8));
        scanf("%d", &pos);
        if (pos >= 1 && pos <= (int)((n + 2) * 8)) {
            size_t byte = (size_t)(pos - 1) / 8;
            int bit = 7 - ((pos - 1) % 8);
            frame[byte] ^= (char)(1U << bit);
        }
    }

    /* Verify by recalculating CRC over original data bytes. */
    if (crc16((unsigned char *)frame, n) == crc &&
        (unsigned char)frame[n] == (unsigned char)(crc >> 8) &&
        (unsigned char)frame[n + 1] == (unsigned char)(crc & 0xFF))
        printf("No error detected\\n");
    else
        printf("Error detected\\n");

    return 0;
}

5. Stop-and-Wait Protocol (simulation)

This is a console simulation of acknowledgements, not a real network transmission.

#include <stdio.h>

int main(void) {
    int frames, i, ack;

    printf("Enter number of frames: ");
    scanf("%d", &frames);

    for (i = 1; i <= frames; ) {
        printf("Sender: sending frame %d\\n", i);
        printf("Was ACK %d received? (1=yes, 0=no): ", i);
        scanf("%d", &ack);

        if (ack == 1) {
            printf("Receiver: ACK %d received.\\n", i);
            i++;
        } else {
            printf("Timeout: retransmitting frame %d.\\n", i);
        }
    }

    printf("All frames transmitted successfully.\\n");
    return 0;
}

6. Sliding Window Protocol (simulation)

This is a console simulation of acknowledgements, not a real network transmission.

#include <stdio.h>

int main(void) {
    int total, window, base = 1, i, ack;

    printf("Enter total number of frames: ");
    scanf("%d", &total);
    printf("Enter window size: ");
    scanf("%d", &window);

    if (window < 1 || total < 1) {
        printf("Invalid input.\\n");
        return 1;
    }

    while (base <= total) {
        int last = base + window - 1;
        if (last > total) last = total;

        printf("\\nSending window: ");
        for (i = base; i <= last; i++) printf("%d ", i);
        printf("\\nEnter highest consecutively acknowledged frame (%d-%d): ",
               base - 1, last);
        scanf("%d", &ack);

        if (ack >= base - 1 && ack <= last) {
            if (ack >= base) base = ack + 1;
            else printf("No new ACK; window stays in place.\\n");
        } else {
            printf("Invalid ACK. Window stays in place.\\n");
        }
    }

    printf("All frames acknowledged.\\n");
    return 0;
}

7. Congestion Control using Leaky Bucket

For each time unit, enter the number of arriving packets; excess packets over bucket capacity are dropped.

#include <stdio.h>

int main(void) {
    int n, capacity, rate, i, packet, stored = 0, sent, dropped;

    printf("Enter bucket capacity: ");
    scanf("%d", &capacity);
    printf("Enter output rate per time unit: ");
    scanf("%d", &rate);
    printf("Enter number of incoming packets/time units: ");
    scanf("%d", &n);

    if (capacity < 1 || rate < 1 || n < 1) {
        printf("Invalid input.\\n");
        return 1;
    }

    for (i = 1; i <= n; i++) {
        printf("Packets arriving at time %d: ", i);
        scanf("%d", &packet);

        stored += packet;
        dropped = 0;
        if (stored > capacity) {
            dropped = stored - capacity;
            stored = capacity;
        }

        sent = stored < rate ? stored : rate;
        stored -= sent;

        printf("Sent: %d, Dropped: %d, Remaining in bucket: %d\\n",
               sent, dropped, stored);
    }

    while (stored > 0) {
        sent = stored < rate ? stored : rate;
        stored -= sent;
        printf("Draining bucket: sent %d, remaining %d\\n", sent, stored);
    }
    return 0;
}
